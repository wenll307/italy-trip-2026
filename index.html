create index if not exists trip_bookings_trip_start_idx on public.trip_bookings (trip_id, start_at); alter table public.trip_bookings enable row level security; -- All members may read bookings for their own trip. drop policy if exists "trip_bookings_select_members_v25" on public.trip_bookings; create policy "trip_bookings_select_members_v25" on public.trip_bookings for select to authenticated using (exists ( select 1 from public.trip_members tm where tm.trip_id = trip_bookings.trip_id and tm.user_id = (select auth.uid()) )); -- Owner and editors may add and edit bookings for their own trip. drop policy if exists "trip_bookings_insert_editors_v25" on public.trip_bookings; create policy "trip_bookings_insert_editors_v25" on public.trip_bookings for insert to authenticated with check ( created_by = (select auth.uid()) and exists (select 1 from public.trip_members tm where tm.trip_id = trip_bookings.trip_id and tm.user_id = (select auth.uid()) and tm.role in ('owner','editor')) ); drop policy if exists "trip_bookings_update_editors_v25" on public.trip_bookings; create policy "trip_bookings_update_editors_v25" on public.trip_bookings for update to authenticated using (exists (select 1 from public.trip_members tm where tm.trip_id = trip_bookings.trip_id and tm.user_id = (select auth.uid()) and tm.role in ('owner','editor'))) with check (exists (select 1 from public.trip_members tm where tm.trip_id = trip_bookings.trip_id and tm.user_id = (select auth.uid()) and tm.role in ('owner','editor'))); drop policy if exists "trip_bookings_delete_editors_v25" on public.trip_bookings; create policy "trip_bookings_delete_editors_v25" on public.trip_bookings for delete to authenticated using (exists (select 1 from public.trip_members tm where tm.trip_id = trip_bookings.trip_id and tm.user_id = (select auth.uid()) and tm.role in ('owner','editor')));
